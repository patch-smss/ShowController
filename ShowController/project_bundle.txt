const listeners = new Map();

const ctx = {
  store,
  channel,
  state: {
    blackout: true,
    intermission: false,
    endAction: 'intermission',
    audioPrefs: { fadeIn: true, fadeOut: true },
    finishedText: null
  },
  on(event, fn) {
    if (!listeners.has(event)) listeners.set(event, new Set());
    listeners.get(event).add(fn);
  },
  emit(event, payload = {}) {
    listeners.get(event)?.forEach(fn => {
      try { fn(payload); } catch (err) { console.error(`Plugin error [${event}]:`, err); }
    });
  },
  actions: {}
};

store.onChange = () => ctx.emit('STORE_UPDATED');

async function pptPost(endpoint, body = {}) {
  const res = await fetch(endpoint, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body)
  });
  return res.json();
}

export async function triggerCue(index) {
  if (index < 0 || index >= store.cues.length) return;
  const cue = store.cues[index];
  ctx.state.finishedText = null;
  store.setActive(cue.id);
  ctx.emit('CUE_TRIGGER', { cue, index });

  if (cue.kind === 'audio') {
    playAudioCue(cue, ctx.state.audioPrefs, () => {
      ctx.state.finishedText = '-- (Finished)';
      ctx.emit('STORE_UPDATED');
    });
    pptPost('/api/ppt/nav', { action: 'focus' }).catch(() => {});
    return;
  }

  if (cue.kind === 'youtube' || cue.kind === 'canva') return;

  if (['ppt', 'image', 'video'].includes(cue.kind)) {
    if (!cue.startSlide) {
      const ins = await pptPost('/api/ppt/insert', { file: cue.src, kind: cue.kind });
      if (ins.ok) { cue.startSlide = ins.startSlide; cue.endSlide = ins.endSlide; }
    }
    if (cue.startSlide) {
      ctx.emit('SET_BLACKOUT', { active: false });
      await pptPost('/api/ppt/nav', { action: 'goto', slide: cue.startSlide });
    }
  }
}

export function nextCue() {
  if (store.isAtEnd) return;
  const idx = store.getActiveIndex();
  if (idx + 1 < store.cues.length) {
    triggerCue(idx + 1);
  } else if (store.cues.length > 0) {
    store.isAtEnd = true;
    ctx.emit('SHOW_ENDED', { endAction: ctx.state.endAction });
  }
}

export function prevCue() {
  const idx = store.getActiveIndex();
  if (store.isAtEnd && idx >= 0) return triggerCue(idx);
  if (idx - 1 >= 0) triggerCue(idx - 1);
}

let activeAudio = null;

function fadeAudio(audioEl, targetVol, durationMs, onDone) {
  if (audioEl._fadeTimer) clearInterval(audioEl._fadeTimer);
  const steps = 25;
  const stepTime = durationMs / steps;
  const volStep = (targetVol - audioEl.volume) / steps;

  audioEl._fadeTimer = setInterval(() => {
    const nextVol = audioEl.volume + volStep;
    if ((volStep > 0 && nextVol >= targetVol) || (volStep < 0 && nextVol <= targetVol)) {
      audioEl.volume = targetVol;
      clearInterval(audioEl._fadeTimer);
      if (onDone) onDone();
    } else {
      audioEl.volume = Math.max(0, Math.min(1, nextVol));
    }
  }, stepTime);
}

export function stopAudio(useFadeOut = true) {
  if (!activeAudio) return;
  const old = activeAudio;
  activeAudio = null;
  if (useFadeOut && !old.paused) {
    fadeAudio(old, 0, 1500, () => { old.pause(); old.remove(); });
  } else {
    if (old._fadeTimer) clearInterval(old._fadeTimer);
    old.pause();
    old.remove();
  }
}

export function playAudioCue(cue, prefs = { fadeIn: true, fadeOut: true }, onEnded = null) {
  stopAudio(prefs.fadeOut);
  const mediaUrl = cue.src.startsWith('blob:')
    ? cue.src
    : '/Media/' + cue.src.split('/').map(encodeURIComponent).join('/');

  const audio = document.createElement('audio');
  audio.src = mediaUrl;
  audio.volume = prefs.fadeIn ? 0 : 1;
  audio.onended = () => {
    if (activeAudio === audio) activeAudio = null;
    audio.remove();
    if (onEnded) onEnded();
  };
  document.body.appendChild(audio);
  audio.play().catch(err => console.warn('Audio playback blocked:', err));
  activeAudio = audio;
  if (prefs.fadeIn) fadeAudio(audio, 1, 1500);
}

const IMG_EXTS = ['png', 'jpg', 'jpeg', 'gif', 'webp', 'svg'];
const AUD_EXTS = ['mp3', 'wav', 'ogg', 'm4a'];
const VID_EXTS = ['mp4', 'mov', 'webm'];

export function detectMediaKind(filename) {
  const ext = filename.split('.').pop().toLowerCase();
  if (IMG_EXTS.includes(ext)) return 'image';
  if (AUD_EXTS.includes(ext)) return 'audio';
  if (VID_EXTS.includes(ext)) return 'video';
  return null;
}

export async function ingestMediaFile(file, store) {
  const kind = detectMediaKind(file.name);
  if (!kind) return;
  if (kind === 'audio') {
    store.addCue(file.name, 'audio', URL.createObjectURL(file));
    return;
  }
  const up = await fetch('/api/upload-media', {
    method: 'POST',
    headers: { 'X-Filename': encodeURIComponent(file.name) },
    body: file
  });
  if (!up.ok) return;

  const ins = await (await fetch('/api/ppt/insert', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ file: file.name, kind })
  })).json();

  store.addCue(file.name, kind, file.name, {
    tempFile: true,
    startSlide: ins.ok ? ins.startSlide : null,
    endSlide: ins.ok ? ins.endSlide : null
  });
}

export const channel = new BroadcastChannel('show_controller_bus');

export const store = {
  cues: [],
  activeId: null,
  isAtEnd: false,
  onChange: null,

  notify() {
    if (typeof this.onChange === 'function') this.onChange();
  },

  getActiveIndex() {
    if (!this.activeId) return -1;
    return this.cues.findIndex(c => c.id === this.activeId);
  },

  addCue(name, kind, src, extra = {}) {
    this.cues.push({ id: crypto.randomUUID(), name, kind, src, ...extra });
    this.notify();
  },

  duplicateCue(index) {
    const orig = this.cues[index];
    if (!orig) return;
    this.cues.push({ ...orig, id: crypto.randomUUID() });
    this.notify();
  },

  removeCue(index) {
    if (this.cues[index]?.id === this.activeId) this.activeId = null;
    this.cues.splice(index, 1);
    this.notify();
  },

  moveCue(fromIndex, toIndex) {
    if (fromIndex === null || fromIndex === toIndex) return;
    const [moved] = this.cues.splice(fromIndex, 1);
    this.cues.splice(toIndex, 0, moved);
    this.notify();
  },

  clearAll() {
    this.cues = [];
    this.activeId = null;
    this.isAtEnd = false;
    this.notify();
  },

  setActive(id) {
    this.activeId = id;
    this.isAtEnd = false;
    this.notify();
  }
};

const webTabs = new Map();

function buildDirectUrl(cue) {
  if (cue.kind === 'youtube') {
    const match = cue.src.match(/(?:youtu\.be\/|youtube\.com\/(?:watch\?v=|embed\/|shorts\/|live\/))([^&?#/]+)/);
    const vidId = match ? match.at(1) : null;
    if (vidId && (cue.end || cue.loop)) {
      let params = 'autoplay=1&rel=0';
      if (cue.start) params += `&start=${parseInt(cue.start, 10)}`;
      if (cue.end) params += `&end=${parseInt(cue.end, 10)}`;
      if (cue.loop) params += `&loop=1&playlist=${vidId}`;
      return `https://youtube.com{vidId}?${params}`;
    }
    let url = cue.src;
    if (cue.start) url += `${url.includes('?') ? '&' : '?'}t=${parseInt(cue.start, 10)}s`;
    return url;
  }
  return cue.src;
}

export function init(container, ctx) {
  const row = document.createElement('div');
  row.className = 'toolbar-row';
  row.innerHTML = `
    <input type="text" id="url-input" placeholder="Paste YouTube or Canva link..." />
    <button id="btn-add-url">+ Add Link</button>
  `;
  container.appendChild(row);

  row.querySelector('#btn-add-url').onclick = () => {
    const input = row.querySelector('#url-input');
    const rawUrl = input.value.trim();
    if (!rawUrl) return;
    const kind = /youtu\.be\/|youtube\.com\//i.test(rawUrl) ? 'youtube' : 'canva';
    ctx.store.addCue(rawUrl, kind, rawUrl);
    input.value = '';
  };

  ctx.on('CUE_PRELOAD', ({ cue }) => {
    if (cue.kind !== 'youtube' && cue.kind !== 'canva') return;
    const tab = window.open(buildDirectUrl(cue), 'tab_' + cue.id);
    if (tab) webTabs.set(cue.id, tab);
    ctx.actions.ensureProjectorOpen().then(() => {
      setTimeout(() => {
        const win = ctx.actions.getProjectorWin();
        if (win && !win.closed) win.focus();
      }, 300);
    });
  });

  ctx.on('CUE_TRIGGER', ({ cue }) => {
    if (cue.kind !== 'youtube' && cue.kind !== 'canva') return;
    let tab = webTabs.get(cue.id);
    if (tab && !tab.closed) tab.focus();
    else {
      tab = window.open(buildDirectUrl(cue), 'tab_' + cue.id);
      if (tab) webTabs.set(cue.id, tab);
    }
  });

  ctx.on('END_SHOW', () => {
    webTabs.forEach(tab => { try { if (tab && !tab.closed) tab.close(); } catch {} });
    webTabs.clear();
  });
}

    const pptRes = await ctx.actions.pptPost('/api/ppt/save', { name });
    const payload = {

      cues: ctx.store.cues.map(({ tempFile, ...rest }) => rest)
    };

    const blob = new Blob([JSON.stringify(payload, null, 2)], { type: 'application/json' });
    const a = document.createElement('a');

    a.download = `${name.replace(/\.json$/i, '')}.json`;
    a.click();

  fileInput.onchange = (e) => {
    const file = e.target.files.item(0);
    if (!file) return;
    const reader = new FileReader();
    reader.onload = async () => {
      try {
        const parsed = JSON.parse(reader.result);
        const cues = Array.isArray(parsed) ? parsed : parsed.cues;
        if (!Array.isArray(cues)) throw new Error();
        if (parsed.masterPptx) {
          await ctx.actions.pptPost('/api/ppt/load', { file: parsed.masterPptx });
        }
        ctx.store.cues = cues;
        ctx.store.activeId = null;
        ctx.store.isAtEnd = false;
        ctx.store.notify();
      } catch {
        alert('Invalid show file.');
      }
    };
    reader.readAsText(file);
    fileInput.value = '';
  };

  container.insertBefore(fileInput, container.firstChild);
  container.insertBefore(loadBtn, container.firstChild);
  container.insertBefore(saveBtn, container.firstChild);

function Handle-MediaApi($rawPath, $req, $res, $mediaDir) {
    if ($rawPath -eq "/api/upload-media" -and $req.HttpMethod -eq "POST") {
        $relPath = [System.Uri]::UnescapeDataString($req.Headers["X-Filename"]).Replace('/', '\').TrimStart('\')
        if ($relPath -and -not $relPath.Contains("..")) {
            $destPath = Join-Path $mediaDir $relPath
            $parent = Split-Path $destPath -Parent
            if (!(Test-Path $parent)) { New-Item -ItemType Directory -Path $parent -Force | Out-Null }
            $fs = [System.IO.File]::Create($destPath)
            $req.InputStream.CopyTo($fs)
            $fs.Close()
            Unblock-File -Path $destPath -ErrorAction SilentlyContinue
            Send-Json $res @{ ok = $true }
        } else {
            $res.StatusCode = 400
            Send-Json $res @{ error = "Invalid path" }

        return $true

    if ($rawPath -eq "/api/cleanup-temp" -and $req.HttpMethod -eq "POST") {
        $body = (New-Object System.IO.StreamReader($req.InputStream)).ReadToEnd() | ConvertFrom-Json

        }
        Send-Json $res @{ ok = $true }
        return $true
    }

    return $false

function Insert-MediaSlide($pres, $fullPath, $kind) {

        $shp.LockAspectRatio = -1
        $scale = [Math]::Min($sw / $shp.Width, $sh / $shp.Height)
        $shp.Width = $shp.Width * $scale
        $shp.Left = ($sw - $shp.Width) / 2; $shp.Top = ($sh - $shp.Height) / 2

    $listener.Prefixes.Add("http://127.0.0.1:$port/")
    $listener.Start()
}

$mediaDir = Join-Path $PSScriptRoot "Media"
if (!(Test-Path $mediaDir)) { New-Item -ItemType Directory -Path $mediaDir | Out-Null }

$mimeTypes = @{
    ".html" = "text/html; charset=utf-8"; ".css" = "text/css; charset=utf-8"; ".js" = "application/javascript; charset=utf-8"
    ".json" = "application/json; charset=utf-8"; ".png" = "image/png"; ".jpg" = "image/jpeg"; ".jpeg" = "image/jpeg"
    ".gif" = "image/gif"; ".webp" = "image/webp"; ".svg" = "image/svg+xml"; ".mp3" = "audio/mpeg"; ".wav" = "audio/wav"
    ".ogg" = "audio/ogg"; ".m4a" = "audio/mp4"; ".mp4" = "video/mp4"; ".mov" = "video/quicktime"
    ".pdf" = "application/pdf";
}

function Send-Json($res, $obj) {
    $bytes = [System.Text.Encoding]::UTF8.GetBytes((ConvertTo-Json -InputObject $obj -Compress))
    $res.ContentType = "application/json; charset=utf-8"
    $res.OutputStream.Write($bytes, 0, $bytes.Length)
    $res.Close()
}

Write-Host "Show Server running on port $port" -ForegroundColor Green

while ($listener.IsListening) {
    try {
        $context = $listener.GetContext()
        $req = $context.Request; $res = $context.Response
        $rawPath = [System.Uri]::UnescapeDataString($req.Url.AbsolutePath)

        if ($rawPath -eq "/api/shutdown") {
            Send-Json $res @{ status = "stopping" }
            if (Get-Command Close-PptCleanly -ErrorAction SilentlyContinue) { Close-PptCleanly }
            Start-Sleep -Milliseconds 400
            DisplaySwitch.exe /clone
            Start-Sleep -Seconds 1
            Get-CimInstance Win32_Process -Filter "Name = 'msedge.exe'" | Where-Object { $_.CommandLine -like "*ShowControllerEdge*" } | ForEach-Object { Stop-Process -Id $_.ProcessId -Force -ErrorAction SilentlyContinue }
            break
        }

        if ($rawPath -eq "/api/plugins") {
            $manifest = @{}
            foreach ($z in $zones) {
                $zPath = Join-Path $pluginsRoot $z
                $files = @(Get-ChildItem -Path $zPath -Filter "*.js" -ErrorAction SilentlyContinue | Sort-Object Name | ForEach-Object { "/plugins/$z/$($_.Name)" })
                $manifest[$z] = $files
            }
            Send-Json $res $manifest
            continue
        }

        if (Get-Command Handle-ScreensApi -ErrorAction SilentlyContinue) { if (Handle-ScreensApi $rawPath $res) { continue } }
        if (Get-Command Handle-PptComApi -ErrorAction SilentlyContinue)  { if (Handle-PptComApi $rawPath $req $res $mediaDir) { continue } }
        if (Get-Command Handle-MediaApi -ErrorAction SilentlyContinue)   { if (Handle-MediaApi $rawPath $req $res $mediaDir) { continue } }
        if (Get-Command Handle-RemoteApi -ErrorAction SilentlyContinue)  { if (Handle-RemoteApi $rawPath $req $res) { continue } }

        if ($rawPath -eq "/") { $rawPath = "/control.html" }
        $rel = $rawPath.TrimStart('/').Replace('/', '\')
        if ($rel.Contains("..")) { $res.StatusCode = 403; $res.Close(); continue }

        $target = Join-Path $PSScriptRoot $rel
        if (Test-Path $target -PathType Leaf) {
            $ext = [System.IO.Path]::GetExtension($target).ToLower()
            $res.ContentType = if ($mimeTypes.ContainsKey($ext)) { $mimeTypes[$ext] } else { "application/octet-stream" }
            $bytes = [System.IO.File]::ReadAllBytes($target)
            $res.ContentLength64 = $bytes.Length
            $res.OutputStream.Write($bytes, 0, $bytes.Length)
        } else { $res.StatusCode = 404 }
        $res.Close()
    } catch {}
}
$listener.Stop()
