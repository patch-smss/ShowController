// Core plugin loader, event bus, and cue execution orchestrator

const listeners = new Map();

const ctx = {
  store,
  channel,
  state: {
    blackout: true,
    intermission: false,
    endAction: 'intermission',
    audioPrefs: { fadeIn: true, fadeOut: true, fadeMs: 1500 },
    settings: { fadeMs: 1500, stopAudioOnVideo: false, aspectRatio: '16:9', imageFit: 'contain', confirmPrompts: true, remotePin: '' },
    finishedText: null
  },
  on(event, fn) {
    if (!listeners.has(event)) listeners.set(event, new Set());
    listeners.get(event).add(fn);
  },
  emit(event, payload = {}) {
    listeners.get(event)?.forEach(fn => {
      try { fn(payload); } catch (err) { console.error(`Plugin error [${event}]:`, err); }
    });
  },
  actions: {}
};

store.onChange = () => ctx.emit('STORE_UPDATED');

async function pptPost(endpoint, body = {}) {
  const res = await fetch(endpoint, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body)
  });
  return res.json();
}

export async function triggerCue(index) {
  if (index < 0 || index >= store.cues.length) return;
  const cue = store.cues[index];
  ctx.state.finishedText = null;
  store.setActive(cue.id);
  ctx.emit('CUE_TRIGGER', { cue, index });

  if (cue.kind === 'audio') {
    playAudioCue(cue, { ...ctx.state.audioPrefs, fadeMs: ctx.state.settings.fadeMs }, () => {
      ctx.state.finishedText = '-- (Finished)';
      ctx.emit('STORE_UPDATED');
    });
    pptPost('/api/ppt/nav', { action: 'focus' }).catch(() => {});
    return;
  }

  if (cue.kind === 'video' && ctx.state.settings.stopAudioOnVideo) {
    stopAudio(ctx.state.audioPrefs.fadeOut, ctx.state.settings.fadeMs);
  }

  if (cue.kind === 'youtube' || cue.kind === 'canva') return;

  if (['ppt', 'image', 'video'].includes(cue.kind)) {
    if (!cue.startSlide) {
      const ins = await pptPost('/api/ppt/insert', { file: cue.src, kind: cue.kind });
      if (ins.ok) { cue.startSlide = ins.startSlide; cue.endSlide = ins.endSlide; }
    }
    if (cue.startSlide) {
      ctx.emit('SET_BLACKOUT', { active: false });
      await pptPost('/api/ppt/nav', { action: 'goto', slide: cue.startSlide });
    }
  }
}

export function nextCue() {
  if (store.cues.length === 0) return;
  if (store.isAtEnd) { store.isAtEnd = false; triggerCue(0); return; }
  const idx = store.getActiveIndex();
  if (idx + 1 < store.cues.length) triggerCue(idx + 1);
  else { store.isAtEnd = true; ctx.emit('SHOW_ENDED', { endAction: ctx.state.endAction }); }
}

export function prevCue() {
  if (store.cues.length === 0) return;
  const idx = store.getActiveIndex();
  if (store.isAtEnd && idx >= 0) { store.isAtEnd = false; triggerCue(idx); return; }
  if (idx <= 0) { triggerCue(store.cues.length - 1); return; }
  triggerCue(idx - 1);
}

let activeAudio = null;

// Hidden Monitor 1 audio player with configurable fade-in and fade-out

function fadeAudio(audioEl, targetVol, durationMs, onDone) {
  if (audioEl._fadeTimer) clearInterval(audioEl._fadeTimer);
  if (!durationMs || durationMs <= 0) {
    audioEl.volume = targetVol;
    if (onDone) onDone();
    return;
  }
  const steps = 25;
  const stepTime = durationMs / steps;
  const volStep = (targetVol - audioEl.volume) / steps;

  audioEl._fadeTimer = setInterval(() => {
    const nextVol = audioEl.volume + volStep;
    if ((volStep > 0 && nextVol >= targetVol) || (volStep < 0 && nextVol <= targetVol)) {
      audioEl.volume = targetVol;
      clearInterval(audioEl._fadeTimer);
      if (onDone) onDone();
    } else {
      audioEl.volume = Math.max(0, Math.min(1, nextVol));
    }
  }, stepTime);
}

export function stopAudio(useFadeOut = true, fadeMs = 1500) {
  if (!activeAudio) return;
  const old = activeAudio;
  activeAudio = null;
  if (useFadeOut && !old.paused) {
    fadeAudio(old, 0, fadeMs, () => { old.pause(); old.remove(); });
  } else {
    if (old._fadeTimer) clearInterval(old._fadeTimer);
    old.pause();
    old.remove();
  }
}

export function playAudioCue(cue, prefs = { fadeIn: true, fadeOut: true, fadeMs: 1500 }, onEnded = null) {
  const ms = prefs.fadeMs ?? 1500;
  stopAudio(prefs.fadeOut, ms);
  const mediaUrl = cue.src.startsWith('blob:')
    ? cue.src
    : '/Media/' + cue.src.split('/').map(encodeURIComponent).join('/');

  const audio = document.createElement('audio');
  audio.src = mediaUrl;
  audio.volume = prefs.fadeIn && ms > 0 ? 0 : 1;
  audio.onended = () => {
    if (activeAudio === audio) activeAudio = null;
    audio.remove();
    if (onEnded) onEnded();
  };
  document.body.appendChild(audio);
  audio.play().catch(err => console.warn('Audio playback blocked:', err));
  activeAudio = audio;
  if (prefs.fadeIn) fadeAudio(audio, 1, ms);
}

const IMG_EXTS = ['png', 'jpg', 'jpeg', 'gif', 'webp', 'svg'];
const AUD_EXTS = ['mp3', 'wav', 'ogg', 'm4a'];
const VID_EXTS = ['mp4', 'mov', 'webm'];

// Media type detector and auto-uploader/inserter for visual files

export function detectMediaKind(filename) {
  const ext = filename.split('.').pop().toLowerCase();
  if (IMG_EXTS.includes(ext)) return 'image';
  if (AUD_EXTS.includes(ext)) return 'audio';
  if (VID_EXTS.includes(ext)) return 'video';
  return null;
}

export async function ingestMediaFile(file, store) {
  const kind = detectMediaKind(file.name);
  if (!kind) return;
  if (kind === 'audio') {
    store.addCue(file.name, 'audio', URL.createObjectURL(file));
    return;
  }
  const up = await fetch('/api/upload-media', {
    method: 'POST',
    headers: { 'X-Filename': encodeURIComponent(file.name) },
    body: file
  });
  if (!up.ok) return;

  const ins = await (await fetch('/api/ppt/insert', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ file: file.name, kind })
  })).json();

  store.addCue(file.name, kind, file.name, {
    tempFile: true,
    startSlide: ins.ok ? ins.startSlide : null,
    endSlide: ins.ok ? ins.endSlide : null
  });
}

// Reactive cue list state container

export const store = {
  cues: [],
  activeId: null,
  isAtEnd: false,
  onChange: null,

  notify() {
    if (typeof this.onChange === 'function') this.onChange();
  },

  getActiveIndex() {
    if (!this.activeId) return -1;
    return this.cues.findIndex(c => c.id === this.activeId);
  },

  addCue(name, kind, src, extra = {}) {
    this.cues.push({ id: crypto.randomUUID(), name, kind, src, ...extra });
    this.notify();
  },

  duplicateCue(index) {
    const orig = this.cues[index];
    if (!orig) return;
    this.cues.push({ ...orig, id: crypto.randomUUID() });
    this.notify();
  },

  removeCue(index) {
    if (this.cues[index]?.id === this.activeId) this.activeId = null;
    this.cues.splice(index, 1);
    this.notify();
  },

  moveCue(fromIndex, toIndex) {
    if (fromIndex === null || fromIndex === toIndex) return;
    const [moved] = this.cues.splice(fromIndex, 1);
    this.cues.splice(toIndex, 0, moved);
    this.notify();
  },

  clearAll() {
    this.cues = [];
    this.activeId = null;
    this.isAtEnd = false;
    this.notify();
  },

  setActive(id) {
    this.activeId = id;
    this.isAtEnd = false;
    this.notify();
  }
};

// Manage YouTube/Canva link cues and external browser tabs
const webTabs = new Map();

function buildDirectUrl(cue) {
  if (cue.kind === 'youtube') {
    const match = cue.src.match(/(?:youtu\.be\/|youtube\.com\/(?:watch\?v=|embed\/|shorts\/|live\/))([^&?#/]+)/);
    const vidId = match ? match.at(1) : null;
    if (vidId && (cue.end || cue.loop)) {
      let params = 'autoplay=1&rel=0';
      if (cue.start) params += `&start=${parseInt(cue.start, 10)}`;
      if (cue.end) params += `&end=${parseInt(cue.end, 10)}`;
      if (cue.loop) params += `&loop=1&playlist=${vidId}`;
      return `https://youtube.com{vidId}?${params}`;
    }
    let url = cue.src;
    if (cue.start) url += `${url.includes('?') ? '&' : '?'}t=${parseInt(cue.start, 10)}s`;
    return url;
  }
  return cue.src;
}

export function init(container, ctx) {
  const row = document.createElement('div');
  row.className = 'toolbar-row';
  row.innerHTML = `
    <input type="text" id="url-input" placeholder="Paste YouTube or Canva link..." />
    <button id="btn-add-url">+ Add Link</button>
  `;
  container.appendChild(row);

  row.querySelector('#btn-add-url').onclick = () => {
    const input = row.querySelector('#url-input');
    const rawUrl = input.value.trim();
    if (!rawUrl) return;
    const kind = /youtu\.be\/|youtube\.com\//i.test(rawUrl) ? 'youtube' : 'canva';
    ctx.store.addCue(rawUrl, kind, rawUrl);
    input.value = '';
  };

  ctx.on('CUE_PRELOAD', ({ cue }) => {
    if (cue.kind !== 'youtube' && cue.kind !== 'canva') return;
    const tab = window.open(buildDirectUrl(cue), 'tab_' + cue.id);
    if (tab) webTabs.set(cue.id, tab);
  });

  ctx.on('CUE_TRIGGER', ({ cue }) => {
    if (cue.kind !== 'youtube' && cue.kind !== 'canva') return;
    let tab = webTabs.get(cue.id);
    if (tab && !tab.closed) tab.focus();
    else {
      tab = window.open(buildDirectUrl(cue), 'tab_' + cue.id);
      if (tab) webTabs.set(cue.id, tab);
    }
  });

  ctx.on('END_SHOW', () => {
    webTabs.forEach(tab => { try { if (tab && !tab.closed) tab.close(); } catch {} });
    webTabs.clear();
  });
}

        ctx.store.isAtEnd = false;
        ctx.store.notify();

        alert('Invalid show file.');
      }
    };

  container.insertBefore(loadBtn, container.firstChild);

# Handle local file uploads (with Windows Unblock-File) and temp file cleanup
function Handle-MediaApi($rawPath, $req, $res, $mediaDir) {
    if ($rawPath -eq "/api/upload-media" -and $req.HttpMethod -eq "POST") {
        $relPath = [System.Uri]::UnescapeDataString($req.Headers["X-Filename"]).Replace('/', '\').TrimStart('\')
        if ($relPath -and -not $relPath.Contains("..")) {
            $destPath = Join-Path $mediaDir $relPath
            $parent = Split-Path $destPath -Parent
            if (!(Test-Path $parent)) { New-Item -ItemType Directory -Path $parent -Force | Out-Null }
            $fs = [System.IO.File]::Create($destPath)
            $req.InputStream.CopyTo($fs)
            $fs.Close()
            Unblock-File -Path $destPath -ErrorAction SilentlyContinue
            Send-Json $res @{ ok = $true }
        } else {
            $res.StatusCode = 400
            Send-Json $res @{ error = "Invalid path" }
        }
        return $true
    }

    if ($rawPath -eq "/api/cleanup-temp" -and $req.HttpMethod -eq "POST") {
        $body = (New-Object System.IO.StreamReader($req.InputStream)).ReadToEnd() | ConvertFrom-Json
        foreach ($f in $body.files) {
            $safe = [System.IO.Path]::GetFileName($f)
            if ($safe) { Remove-Item (Join-Path $mediaDir $safe) -Force -ErrorAction SilentlyContinue }
        }
        Send-Json $res @{ ok = $true }
        return $true
    }

    return $false
}

    }

        if ($p.FullName -eq $masterPath) { $script:masterPres = $p; break }

    if ($null -eq $script:masterPres) {

# Slide insertion for .pptx, images, videos, and hidden intermission slide

function Insert-MediaSlide($pres, $fullPath, $kind) {
    $idx = $pres.Slides.Count + 1
    $sl = $pres.Slides.Add($idx, 12)
    $sl.FollowMasterBackground = 0; $sl.Background.Fill.Solid(); $sl.Background.Fill.ForeColor.RGB = 0
    $sw = $pres.PageSetup.SlideWidth; $sh = $pres.PageSetup.SlideHeight
    $cfg = if (Get-Command Get-ShowSettings -ErrorAction SilentlyContinue) { Get-ShowSettings } else { @{ imageFit = "contain" } }

    if ($kind -eq "image") {
        $shp = $sl.Shapes.AddPicture($fullPath, 0, -1, 0, 0)
        $shp.LockAspectRatio = -1
        $scale = if ($cfg.imageFit -eq "cover") { [Math]::Max($sw / $shp.Width, $sh / $shp.Height) } else { [Math]::Min($sw / $shp.Width, $sh / $shp.Height) }
        $shp.Width = $shp.Width * $scale
        $shp.Left = ($sw - $shp.Width) / 2; $shp.Top = ($sh - $shp.Height) / 2
    } elseif ($kind -eq "video") {
        $shp = $sl.Shapes.AddMediaObject2($fullPath, 0, -1, 0, 0, $sw, $sh)
        $shp.AnimationSettings.PlaySettings.PlayOnEntry = -1
        $shp.AnimationSettings.PlaySettings.PauseAnimation = 0
        $shp.AnimationSettings.PlaySettings.HideWhileNotPlaying = -1
    }
    return $idx
}

function Ensure-IntermissionSlide($pres, $mediaDir, $imgFile) {
    if ($script:intermissionSlide -gt 0 -and $script:intermissionSlide -le $pres.Slides.Count) {
        $sl = $pres.Slides.Item($script:intermissionSlide)
        $sl.SlideShowTransition.Hidden = -1
        if (-not $imgFile) { return $script:intermissionSlide }
        while ($sl.Shapes.Count -gt 0) { $sl.Shapes.Item(1).Delete() }
    } else {
        $idx = $pres.Slides.Count + 1
        $sl = $pres.Slides.Add($idx, 12)
        $sl.SlideShowTransition.Hidden = -1
        $script:intermissionSlide = $idx
    }
    $sl.FollowMasterBackground = 0; $sl.Background.Fill.Solid(); $sl.Background.Fill.ForeColor.RGB = 0
    $sw = $pres.PageSetup.SlideWidth; $sh = $pres.PageSetup.SlideHeight
    if ($imgFile) {
        $full = Join-Path $mediaDir ($imgFile.Replace('/', '\').TrimStart('\'))
        if (Test-Path $full) {
            $shp = $sl.Shapes.AddPicture($full, 0, -1, 0, 0)
            $shp.LockAspectRatio = -1
            $scale = [Math]::Min($sw / $shp.Width, $sh / $shp.Height)
            $shp.Width = $shp.Width * $scale
            $shp.Left = ($sw - $shp.Width) / 2; $shp.Top = ($sh - $shp.Height) / 2
            return $script:intermissionSlide
        }
    }
    $tb = $sl.Shapes.AddTextbox(1, 0, ($sh / 2) - 40, $sw, 80)
    $tb.TextFrame.TextRange.Text = "INTERMISSION"
    $tb.TextFrame.TextRange.Font.Size = 44
    $tb.TextFrame.TextRange.Font.Color.RGB = 0x888888
    $tb.TextFrame.TextRange.ParagraphFormat.Alignment = 2
    return $script:intermissionSlide
}

# Live slideshow sync and COM navigation (next, prev, goto, blackout, freeze)

function Handle-PptNavApi($rawPath, $req, $res, $mediaDir) {
    if ($rawPath -eq "/api/ppt/sync") {
        try {
            $app = Get-PptApp; $pres = Ensure-MasterPresentation $mediaDir; $list = @()
            for ($i = 1; $i -le $pres.Slides.Count; $i++) {
                $sl = $pres.Slides.Item($i)
                $title = if ($i -eq 1) { "Stage Blackout (Slide 1)" } elseif ($sl.Shapes.HasTitle -eq -1) { $sl.Shapes.Title.TextFrame.TextRange.Text } else { "Slide $i" }
                $list += @{ index = $i; title = $title.Trim(); hidden = ($sl.SlideShowTransition.Hidden -eq -1) }
            }
            $hasWin = ($app.SlideShowWindows.Count -gt 0)
            Send-Json $res @{
                ok = $true; frozen = ($script:frozenTempSlide -gt 0); slides = $list
                current = if ($hasWin) { $app.SlideShowWindows.Item(1).View.CurrentShowPosition } else { 0 }
                state = if ($hasWin) { [int]$app.SlideShowWindows.Item(1).View.State } else { 0 }
            }
        } catch { Send-Json $res @{ ok = $false; error = $_.Exception.Message } }
        return $true
    }

    if ($rawPath -eq "/api/ppt/nav" -and $req.HttpMethod -eq "POST") {
        try {
            $body = (New-Object System.IO.StreamReader($req.InputStream)).ReadToEnd() | ConvertFrom-Json
            $pres = Ensure-MasterPresentation $mediaDir; $ssWin = Start-StageSlideShow $pres; $view = $ssWin.View
            switch ($body.action) {
                "focus"    { $ssWin.Activate() }
                "next"     { $view.State = 1; $view.Next() }
                "prev"     { $view.State = 1; $view.Previous() }
                "goto"     { $view.State = 1; $view.GotoSlide([int]$body.slide) }
                "blackout" { $view.State = if ($body.active) { 3 } else { 1 } }
                "freeze"   {
                    if ($body.active -and $script:frozenTempSlide -eq 0) {
                        $dup = $pres.Slides.Item($view.CurrentShowPosition).Duplicate()
                        $script:frozenTempSlide = $dup.SlideIndex; $view.GotoSlide($script:frozenTempSlide)
                    } elseif (-not $body.active -and $script:frozenTempSlide -gt 0) {
                        $view.GotoSlide([Math]::Max(1, $script:frozenTempSlide - 1))
                        $pres.Slides.Item($script:frozenTempSlide).Delete(); $script:frozenTempSlide = 0
                    }
                }
            }
            Send-Json $res @{ ok = $true; current = $view.CurrentShowPosition; state = [int]$view.State; frozen = ($script:frozenTempSlide -gt 0) }
        } catch { $res.StatusCode = 500; Send-Json $res @{ ok = $false; error = $_.Exception.Message } }
        return $true
    }

    return $false
}

function Test-RemotePin($providedPin) {

function Handle-RemoteApi($rawPath, $req, $res, $mediaDir) {
    if ($rawPath -eq "/api/phone-cmd" -and $req.HttpMethod -eq "POST") {
        $cmdObj = (New-Object System.IO.StreamReader($req.InputStream)).ReadToEnd() | ConvertFrom-Json
        if (-not (Test-RemotePin $cmdObj.pin)) {
            $res.StatusCode = 401; Send-Json $res @{ ok = $false; error = "Invalid PIN" }; return $true
        }
        [void]$script:phoneQueue.Add($cmdObj)
        Send-Json $res @{ ok = $true }
        return $true
    }

    if ($rawPath -eq "/api/phone-upload" -and $req.HttpMethod -eq "POST") {
        if (-not (Test-RemotePin $req.Headers["X-Pin"])) {
            $res.StatusCode = 401; Send-Json $res @{ ok = $false; error = "Invalid PIN" }; return $true
        }
        $safeName = [System.IO.Path]::GetFileName([System.Uri]::UnescapeDataString($req.Headers["X-Filename"]))
        if (-not $safeName) { $res.StatusCode = 400; Send-Json $res @{ ok = $false; error = "Missing filename" }; return $true }
        $destPath = Join-Path $mediaDir $safeName
        $fs = [System.IO.File]::Create($destPath)
        $req.InputStream.CopyTo($fs)
        $fs.Close()
        Unblock-File -Path $destPath -ErrorAction SilentlyContinue
        $playNow = ($req.Headers["X-PlayNow"] -eq "1")
        [void]$script:phoneQueue.Add(@{ type = "ADD_MEDIA"; file = $safeName; playNow = $playNow })
        Send-Json $res @{ ok = $true; file = $safeName }
        return $true
    }

    if ($rawPath -eq "/api/poll") {
        $items = @()
        while ($script:phoneQueue.Count -gt 0) {
            $items += $script:phoneQueue[0]
            $script:phoneQueue.RemoveAt(0)
        }
        $json = if ($items.Count -eq 0) { "[]" } else { ConvertTo-Json -InputObject @($items) -Compress }
        $bytes = [System.Text.Encoding]::UTF8.GetBytes($json)
        $res.ContentType = "application/json; charset=utf-8"
        $res.OutputStream.Write($bytes, 0, $bytes.Length)
        $res.Close()
        return $true
    }

    return $false
}

        $_.MainWindowHandle -ne 0 -and $_.MainWindowTitle -like "*Show Control Deck*"
    } | ForEach-Object {
        [Win32Window]::ShowWindow($_.MainWindowHandle, 9) | Out-Null

    }
}

                    next = [int]$saved.hotkeys.next
                    prev = [int]$saved.hotkeys.prev
                    blackout = [int]$saved.hotkeys.blackout
                }
            }
        } catch {}
    }
    return $defaults
}

function Handle-SettingsApi($rawPath, $req, $res) {
    if ($rawPath -eq "/api/settings") {
        if ($req.HttpMethod -eq "POST") {
            $raw = (New-Object System.IO.StreamReader($req.InputStream)).ReadToEnd()
            $obj = $raw | ConvertFrom-Json
            $obj | ConvertTo-Json -Depth 4 | Out-File -FilePath $script:settingsPath -Encoding utf8
            if ($null -ne $obj.hotkeys -and [Type]::GetType("GlobalHotkeyHost")) {
                try { [GlobalHotkeyHost]::Rebind([uint32]$obj.hotkeys.next, [uint32]$obj.hotkeys.prev, [uint32]$obj.hotkeys.blackout) } catch {}
            }
            Send-Json $res @{ ok = $true }
            return $true
        }
        Send-Json $res (Get-ShowSettings)
        return $true
    }
    return $false
}
